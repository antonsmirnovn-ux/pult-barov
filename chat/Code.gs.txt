function setup() {
  let id = P.getProperty('DB_ID');
  let ss = id ? SpreadsheetApp.openById(id) : null;
  if (!ss) {
    ss = SpreadsheetApp.create('Рабочий чат');
    P.setProperty('DB_ID', ss.getId());
  }
  const need = { msgs: M_COLS, users: U_COLS, push: ['uid', 'fcm', 'ts'], bookings: B_COLS };
  Object.keys(need).forEach(function (n) {
    let s = ss.getSheetByName(n);
    if (!s) s = ss.insertSheet(n);
    if (!s.getLastRow()) s.appendRow(need[n]);
    else if (s.getLastColumn() < need[n].length) s.getRange(1, 1, 1, need[n].length).setValues([need[n]]);
    s.setFrozenRows(1);
  });
  const def = ss.getSheetByName('Sheet1') || ss.getSheetByName('Лист1');
  if (def && ss.getSheets().length > 1) ss.deleteSheet(def);
  ss.getSheetByName('users').getRange('C:C').setNumberFormat('@');
  ss.getSheetByName('msgs').getRange('H:L').setNumberFormat('@');
  ss.getSheetByName('bookings').getRange('B:I').setNumberFormat('@'); // текст, даты «13.11» и т.п. не превращаются в числа
  // первый админ — владелец. PIN генерируется случайно и печатается в журнал выполнения.
  const u = ss.getSheetByName('users');
  if (u.getLastRow() < 2) {
    const pin = String(10000000 + Math.floor(Math.random() * 90000000));
    u.appendRow([1, 'Админ', pin, 'both', true, '', true]);
    Logger.log('PIN первого админа: ' + pin + ' — смените его в приложении (Ещё → Сотрудники).');
  }
  folder_();
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'cron') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('cron').timeBased().everyMinutes(5).create();
  // первый доступ к таблице «заказ пива» (чтобы Google спросил разрешение сразу)
  SpreadsheetApp.openById(DELA_SHEET_ID).getSheetByName(DELA_TAB).getLastRow();
  Logger.log('Готово. Таблица: ' + ss.getUrl());
}

/* ================= HTTP ================= */
function doGet(e) { return out_({ ok: true, data: 'Рабочий чат: сервер работает', sectionAccessVersion: 1, staffPasswordVersion: 1 }); }

function doPost(e) {
  let res;
  try {
    const b = JSON.parse((e.postData && e.postData.contents) || '{}');
    res = { ok: true, data: route_(b) };
  } catch (err) {
    res = { ok: false, error: String(err && err.message || err) };
  }
  return out_(res);
}
function out_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function route_(b) {
  if (b.op === 'login') return login_(b.pin);
  const me = auth_(b.token);
  switch (b.op) {
    case 'sync': return sync_(me, b);
    case 'post': return post_(me, b);
    case 'photo': return photo_(me, b.id);
    case 'edit': return editMsg_(me, b);
    case 'del': return delMsg_(me, b.id);
    case 'push_reg': return pushReg_(me, b.fcm);
    case 'push_unreg': return pushUnreg_(me, b.fcm);
    case 'push_test': sendTo_([me.id], 'Рабочий чат', 'Уведомления работают 👍', APP_URL); return true;
    case 'mute': return muteSet_(me, b.mute);
    case 'access_save': admin_(me); protectOwner_(me, b.user && b.user.id); return saveSectionAccess_(b.user);
    case 'users': admin_(me); return listUsers_();
    case 'user_save': admin_(me); return userSave_(me, b.user);
    case 'user_del': admin_(me); return userDel_(b.id);
    case 'staff_password_read': owner_(me); return readStaffPassword_(b.id);
    case 'staff_password_save': owner_(me); return saveStaffPassword_(me, b.id, b.password);
    case 'book_list': return bookList_(me, b);
    case 'book_save': return bookSave_(me, b.book);
    case 'book_del': return bookDel_(me, b.id, b.bar, b.tlabel);
    case 'zp_keys': return zpKeys_(me);
  }
  throw new Error('bad_op');
}

/* ================= пользователи ================= */
function rows_(s, cols) {
  const last = s.getLastRow();
  if (last < 2) return [];
  return s.getRange(2, 1, last - 1, cols.length).getValues().map(function (v, i) {
    const o = { _row: i + 2 };
    cols.forEach(function (c, j) { o[c] = v[j]; });
    return o;
  });
}
function users_(fresh) {
  const c = CacheService.getScriptCache(), hit = !fresh && c.get('users-v2');
  if (hit) return JSON.parse(hit);
  const list = rows_(sh_('users'), U_COLS).filter(function (u) { return u.id !== '' && u.active !== false; })
    .map(function (u) { return { id: +u.id, name: String(u.name), pin: String(u.pin), bars: barsOf_(u.bars), admin: u.admin === true, helper: String(u.admin) === 'helper', zp: u.zp === true, token: String(u.token || ''), mute: muteOf_(u.mute), access: readSectionAccess_(u.access), _row: u._row }; });
  if (!fresh) c.put('users-v2', JSON.stringify(list.map(pub_)), 60);
  return list;
}
function barsOf_(v) { v = String(v || '').toLowerCase(); return v === 'both' || v === 'оба' ? ['13', 'epo'] : (v.indexOf('epo') >= 0 || v.indexOf('епо') >= 0 ? ['epo'] : ['13']); }
function muteOf_(v) { return String(v || '').split(',').filter(function (k) { return MUTE_KINDS.indexOf(k) >= 0; }); }
function pub_(u) { return { id: u.id, name: u.name, bars: u.bars, admin: u.admin, owner: isOwner_(u), helper: u.helper, zp: u.zp || u.admin, mute: u.mute || [], access: u.access }; }
function dropCache_() { CacheService.getScriptCache().removeAll(['users', 'users-v2']); }

const ACCESS_MODE = ['none', 'read', 'write'];
function checkedSectionAccess_(value) {
  if (!value || ACCESS_MODE.indexOf(value.issues) < 0 || ACCESS_MODE.indexOf(value.booking) < 0)
    throw new Error('Выберите права для обоих разделов');
  return { issues: value.issues, booking: value.booking };
}
function readSectionAccess_(value) {
  if (!value) return { issues: 'write', booking: 'write' };
  try { return checkedSectionAccess_(JSON.parse(String(value))); }
  catch (_) { return { issues: 'none', booking: 'none' }; }
}
function hasSection_(u, section, write) {
  const mode = (u.access || readSectionAccess_(''))[section];
  return mode === 'write' || (!write && mode === 'read');
}
function hasBarSection_(u, bar, section, write) {
  return u.bars.indexOf(String(bar)) >= 0 && hasSection_(u, section, write);
}
function requireBarSection_(u, bar, section, write) {
  if (!hasBarSection_(u, bar, section, write)) throw new Error('forbidden');
}

function saveSectionAccess_(change) {
  const access = checkedSectionAccess_(change && change.access);
  const id = Number(change.id);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const sheet = sh_('users');
    const user = rows_(sheet, U_COLS).find(function (entry) {
      return Number(entry.id) === id && entry.id !== '' && entry.active !== false;
    });
    if (!user) throw new Error('Сотрудник не найден');
    const column = U_COLS.indexOf('access') + 1;
    const header = sheet.getRange(1, column);
    if (header.getValue() === '') header.setValue('access');
    else if (header.getValue() !== 'access') throw new Error('Название колонки прав доступа не совпадает');
    sheet.getRange(user._row, column).setValue(JSON.stringify(access));
    SpreadsheetApp.flush();
    dropCache_();
    return listUsers_();
  } finally { lock.releaseLock(); }
}

// Защита от подбора PIN: после 10 неудачных попыток за 15 минут вход закрывается на 15 минут для всех.
const LOGIN_MAX_FAILS = 10, LOGIN_WINDOW_S = 900;
function loginFails_() { return +(CacheService.getScriptCache().get('lfail') || 0); }
function loginFail_() {
  const c = CacheService.getScriptCache(), lock = LockService.getScriptLock();
  let n;
  if (lock.tryLock(3000)) { try { n = loginFails_() + 1; c.put('lfail', String(n), LOGIN_WINDOW_S); } finally { lock.releaseLock(); } }
  else { n = loginFails_() + 1; c.put('lfail', String(n), LOGIN_WINDOW_S); }
  if (n === LOGIN_MAX_FAILS) {
    try { sendTo_(users_().filter(function (u) { return u.admin; }).map(function (u) { return u.id; }), '🔒 Рабочий чат', 'Много неверных PIN подряд — вход закрыт на 15 минут', APP_URL); } catch (e) {}
  }
}

function login_(pin) {
  pin = String(pin || '').trim();
  if (!pin) throw new Error('bad_pin');
  if (loginFails_() >= LOGIN_MAX_FAILS) throw new Error('locked');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const tag = passwordTag_(pin, P.getProperty(PASSWORD_KEY));
    const u = users_(true).find(function (x) { return passwordMatches_(x.pin, pin, tag); });
    if (u) {
      let token = u.token;
      if (!token) {
        token = Utilities.getUuid().replace(/-/g, '');
        sh_('users').getRange(u._row, U_COLS.indexOf('token') + 1).setValue(token);
        SpreadsheetApp.flush();
        dropCache_();
      }
      return { token: token, me: pub_(u) };
    }
  } finally { lock.releaseLock(); }
  loginFail_();
  Utilities.sleep(1500);
  throw new Error('bad_pin');
}
function auth_(tok) {
  const u = tok && users_(true).filter(function (x) { return x.token && x.token === tok; })[0];
  if (!u) throw new Error('auth');
  return u;
}
function admin_(me) { if (!me.admin) throw new Error('forbidden'); }

// setup() creates account 1 for the owner. The admin role alone does not grant access to passwords.
const OWNER_ID = 1;
function isOwner_(user) { return user.id === OWNER_ID && user.admin === true; }
function owner_(me) { if (!isOwner_(me)) throw new Error('forbidden'); }
function protectOwner_(me, id) { if (Number(id) === OWNER_ID) owner_(me); }

function listUsers_() {
  return rows_(sh_('users'), U_COLS).filter(function (u) { return u.id !== '' && u.active !== false; }).map(function (u) {
    return pub_({ id: +u.id, name: String(u.name), bars: barsOf_(u.bars), admin: u.admin === true, helper: String(u.admin) === 'helper', zp: u.zp === true, mute: muteOf_(u.mute), access: readSectionAccess_(u.access) });
  });
}
function userSave_(me, x) {
  if (!x) throw new Error('staff_not_found');
  const name = String(x.name || '').trim(), password = String(x.pin || '').trim();
  protectOwner_(me, x.id);
  const bars = x.bars && x.bars.length === 2 ? 'both' : (x.bars && x.bars[0] === 'epo' ? 'epo' : '13');
  // роль: true — админ (управляет сотрудниками), 'helper' — помощник (уведомления как у админа), false — сотрудник
  const role = x.role === 'admin' ? true : x.role === 'helper' ? 'helper' : false;
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const s = sh_('users'), all = rows_(s, U_COLS);
    const cur = x.id ? activeStaffRow_(all, x.id) : null;
    if (cur) {
      if (!name && x.name !== undefined) throw new Error('Нужно имя');
      if ((+cur.id === me.id || +cur.id === OWNER_ID) && role !== true) throw new Error('Нельзя снять админа с самого себя');
      let sealed = cur.pin, token = cur.token;
      if (password) {
        owner_(auth_(me.token));
        if (!passwordMatches_(cur.pin, password, passwordTag_(password, P.getProperty(PASSWORD_KEY)))) {
          sealed = newStaffPassword_(all, Number(cur.id), password);
          token = '';
        }
      }
      s.getRange(cur._row, 2, 1, 5).setValues([[name || cur.name, sealed, bars, role, token]]);
      s.getRange(cur._row, 8).setValue(!!x.zp);
    } else {
      owner_(auth_(me.token));
      if (!name) throw new Error('Нужно имя');
      const id = all.reduce(function (m, u) { return Math.max(m, +u.id || 0); }, 0) + 1;
      s.appendRow([id, name, newStaffPassword_(all, id, password), bars, role, '', true, !!x.zp]);
    }
    SpreadsheetApp.flush();
    dropCache_();
  } finally { lock.releaseLock(); }
  return listUsers_();
}
function userDel_(id) {
  if (Number(id) === OWNER_ID) throw new Error('forbidden');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const s = sh_('users'), u = activeStaffRow_(rows_(s, U_COLS), id);
    s.getRange(u._row, 6, 1, 2).setValues([['', false]]);
    SpreadsheetApp.flush();
    dropCache_();
  } finally { lock.releaseLock(); }
  return listUsers_();
}

/* ================= пароли сотрудников ================= */
const PASSWORD_KEY = 'STAFF_PASSWORD_KEY_V1';
const PASSWORD_FORMAT = 'pw1:';

function activeStaffRow_(rows, id) {
  const user = rows.find(function (row) { return row.id !== '' && Number(row.id) === Number(id) && row.active !== false; });
  if (!user) throw new Error('staff_not_found');
  return user;
}
function passwordBytes_(value) { return Utilities.newBlob(String(value)).getBytes(); }
function passwordEncode_(bytes) { return Utilities.base64EncodeWebSafe(Array.from(bytes, function (b) { return b << 24 >> 24; })); }
function passwordDecode_(encoded) { return Uint8Array.from(Utilities.base64DecodeWebSafe(encoded)); }
function passwordMac_(master, label) {
  return Utilities.computeHmacSha256Signature(passwordBytes_(label), Utilities.base64DecodeWebSafe(master));
}
function passwordTag_(password, master) {
  return master ? passwordEncode_(passwordMac_(master, 'staff password lookup\n' + password)) : '';
}
function passwordMatches_(stored, password, tag) {
  const value = String(stored);
  if (!value.startsWith(PASSWORD_FORMAT)) return value === password;
  if (!tag) return false;
  const saved = value.split(':')[1];
  if (saved.length !== tag.length) return false;
  let difference = 0;
  for (let i = 0; i < tag.length; i++) difference |= tag.charCodeAt(i) ^ saved.charCodeAt(i);
  return difference === 0;
}
function passwordRandom_() {
  const seed = Utilities.getUuid() + Utilities.getUuid();
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, passwordBytes_(seed));
}
function passwordCipher_(master, nonce, id) {
  const key = Uint8Array.from(passwordMac_(master, 'staff password encryption'));
  return StaffPasswordCipher.xchacha20poly1305(key, nonce, Uint8Array.from(passwordBytes_(id)));
}
function newStaffPassword_(rows, id, password) {
  if (!/^[A-Za-z0-9]{8,32}$/.test(password) || (password.match(/[A-Za-z]/g) || []).length < 4 || (password.match(/\d/g) || []).length < 4)
    throw new Error('staff_password_format');
  let master = P.getProperty(PASSWORD_KEY);
  if (!master) {
    if (rows.some(function (row) { return String(row.pin).startsWith(PASSWORD_FORMAT); }))
      throw new Error('staff_password_unavailable');
    master = passwordEncode_(passwordRandom_());
    P.setProperty(PASSWORD_KEY, master);
  }
  const tag = passwordTag_(password, master);
  if (rows.some(function (row) { return row.active !== false && Number(row.id) !== id && passwordMatches_(row.pin, password, tag); }))
    throw new Error('staff_password_taken');
  const nonce = Uint8Array.from(passwordRandom_()).slice(0, 24);
  const encrypted = passwordCipher_(master, nonce, id).encrypt(Uint8Array.from(passwordBytes_(password)));
  return PASSWORD_FORMAT + tag + ':' + passwordEncode_(nonce) + ':' + passwordEncode_(encrypted);
}
function readStaffPassword_(id) {
  const user = activeStaffRow_(rows_(sh_('users'), U_COLS), id);
  const stored = String(user.pin);
  if (!stored.startsWith(PASSWORD_FORMAT)) return stored;
  try {
    const parts = stored.split(':');
    const master = P.getProperty(PASSWORD_KEY);
    const bytes = passwordCipher_(master, passwordDecode_(parts[2]), Number(user.id)).decrypt(passwordDecode_(parts[3]));
    return String.fromCharCode.apply(null, bytes);
  } catch (_) { throw new Error('staff_password_unavailable'); }
}
function saveStaffPassword_(me, id, value) {
  const password = typeof value === 'string' ? value.trim() : '';
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    owner_(auth_(me.token));
    const sheet = sh_('users'), rows = rows_(sheet, U_COLS);
    const user = activeStaffRow_(rows, id);
    if (String(user.pin).startsWith(PASSWORD_FORMAT) && passwordMatches_(user.pin, password, passwordTag_(password, P.getProperty(PASSWORD_KEY))))
      return { changed: false };
    const sealed = newStaffPassword_(rows, Number(user.id), password);
    sheet.getRange(user._row, U_COLS.indexOf('pin') + 1, 1, 4).setValues([[sealed, user.bars, user.admin, '']]);
    SpreadsheetApp.flush();
    dropCache_();
    return { changed: true };
  } finally { lock.releaseLock(); }
}

/* ================= сообщения ================= */
function verGet_() { return +(CacheService.getScriptCache().get('ver') || P.getProperty('VER') || 0); }
function verBump_() {
  const v = Date.now();
  P.setProperty('VER', String(v));
  CacheService.getScriptCache().put('ver', String(v), 21600);
  return v;
}

function sync_(me, b) {
  maybeStatuses_();
  const since = +b.since || 0, ver = verGet_();
  const res = { me: pub_(me), ver: ver, msgs: [] };
  if (b.withUsers && me.admin) res.people = listUsers_();
  if (!hasSection_(me, 'issues', false)) return res;
  if (since && since >= ver) return res;
  const s = sh_('msgs'), last = s.getLastRow();
  if (last < 2) return res;
  const from = Math.max(2, last - (since ? 3000 : 1500) + 1);
  const vals = s.getRange(from, 1, last - from + 1, M_COLS.length).getValues();
  vals.forEach(function (v) {
    if (+v[2] <= since) return;
    if (me.bars.indexOf(String(v[3])) < 0) return;
    if (!CH[String(v[4])]) return; // старые разделы (например, «Смены») не отдаём
    res.msgs.push(msg_(v));
  });
  return res;
}
function msg_(v) {
  const o = {}; M_COLS.forEach(function (c, j) { o[c] = v[j]; });
  o.id = +o.id; o.upd = +o.upd; o.uid = +o.uid;
  o.ts = o.ts instanceof Date ? o.ts.toISOString() : new Date(+o.ts).toISOString();
  o.photo = String(o.photo || ''); o.text = String(o.text || ''); o.tag = String(o.tag || ''); o.status = String(o.status || ''); o.note = String(o.note || ''); o.cid = String(o.cid || '');
  let files = [];
  try { files = JSON.parse(String(o.files || '[]')) || []; } catch (e) { files = []; }
  if (!files.length && o.photo) files = [{ id: o.photo, name: 'фото.jpg', type: 'image/jpeg' }]; // старые записи с одним фото
  o.files = files;
  if (o.status === 'del') { o.text = ''; o.photo = ''; o.files = []; }
  return o;
}

// Повторная отправка того же сообщения (тот же cid) не создаёт дубль: возвращаем уже сохранённое.
function findByCid_(s, cid) {
  if (!cid) return null;
  const hit = s.getRange(1, M_COLS.indexOf('cid') + 1, s.getMaxRows(), 1).createTextFinder(cid).matchEntireCell(true).findNext();
  return hit ? s.getRange(hit.getRow(), 1, 1, M_COLS.length).getValues()[0] : null;
}

/** Сохраняет вложения на Диск. list: [{name, data:'data:<type>;base64,...'}]. Возвращает [{id,name,type,size}]. */
function saveFiles_(me, bar, list) {
  list = Array.isArray(list) ? list : [];
  if (list.length > MAX_FILES) throw new Error('Не больше ' + MAX_FILES + ' вложений');
  if (!list.length) return [];
  const now = new Date();
  const f = subFolder_(subFolder_(folder_(), BARS[bar]), Utilities.formatDate(now, TZ, 'yyyy-MM'));
  return list.map(function (x, i) {
    const m = /^data:([\w.+-]+\/[\w.+-]+)?(?:;[^,]*)?;base64,(.+)$/.exec(String(x && x.data || ''));
    if (!m) throw new Error('bad_photo');
    const bytes = Utilities.base64Decode(m[2]);
    if (bytes.length > MAX_FILE_MB * 1024 * 1024) throw new Error('Файл больше ' + MAX_FILE_MB + ' МБ');
    const type = m[1] || 'application/octet-stream';
    let name = String(x.name || '').replace(/[\\\/:*?"<>|]/g, '_').slice(0, 80) || ('файл ' + (i + 1));
    const fname = 'Проблема ' + Utilities.formatDate(now, TZ, 'dd.MM.yyyy HH-mm') + ' ' + me.name + ' — ' + name;
    const id = f.createFile(Utilities.newBlob(bytes, type, fname)).getId();
    return { id: id, name: name, type: type, size: bytes.length };
  });
}

function post_(me, b) {

/** Правка проблемы: текст, удаление части вложений, новые вложения. Строка в «списке дел» обновляется (цвет не трогаем). */
function editMsg_(me, b) {
  const s = sh_('msgs');
  let r = findMsgRow_(s, b.id), v = s.getRange(r, 1, 1, M_COLS.length).getValues()[0];
  if (!canChange_(me, v)) throw new Error(+v[5] === me.id ? 'closed' : 'forbidden');
  const text = String(b.text == null ? v[7] : b.text).trim().slice(0, 2000);
  const cur = msg_(v).files, keep = Array.isArray(b.keep) ? b.keep.map(String) : cur.map(function (f) { return f.id; });
  const add = Array.isArray(b.add) ? b.add : [];
  if (cur.filter(function (f) { return keep.indexOf(f.id) >= 0; }).length + add.length > MAX_FILES) throw new Error('Не больше ' + MAX_FILES + ' вложений');
  const added = saveFiles_(me, String(v[3]), add);
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    r = findMsgRow_(s, b.id); v = s.getRange(r, 1, 1, M_COLS.length).getValues()[0];
    if (!canChange_(me, v)) throw new Error('closed');
    const files = msg_(v).files.filter(function (f) { return keep.indexOf(f.id) >= 0; }).concat(added);
    if (!text && !files.length) throw new Error('empty');
    const firstImg = (files.filter(function (f) { return /^image\//.test(f.type); })[0] || {}).id || '';
    const upd = bumpRow_(s, r);
    s.getRange(r, 8, 1, 2).setValues([[text, firstImg]]);
    s.getRange(r, M_COLS.indexOf('files') + 1).setValue(JSON.stringify(files));
    if (v[9]) {
      try { const f = delaFind_(v[9]); f.cells.forEach(function (c) { f.d.getRange(c.getRow(), 2).setValue(delaLine_(String(v[3]), String(v[6]), text, files, v[9])); }); }
      catch (e) { console.warn('dela edit', e); }
    }
    SpreadsheetApp.flush();
    P.setProperty('VER', String(upd)); CacheService.getScriptCache().put('ver', String(upd), 21600);
    v = s.getRange(r, 1, 1, M_COLS.length).getValues()[0];
  } finally { lock.releaseLock(); }
  return msg_(v);
}

/** Удаление проблемы. Она убирается и из «списка дел». */
function delMsg_(me, id) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  let v, r;
  try {
    const s = sh_('msgs');
    r = findMsgRow_(s, id); v = s.getRange(r, 1, 1, M_COLS.length).getValues()[0];
    if (v[10] === 'del') { requireBarSection_(me, v[3], 'issues', true); return msg_(v); }

    const upd = bumpRow_(s, r);
    s.getRange(r, 11).setValue('del');
    if (v[9]) {
      try { const f = delaFind_(v[9]); f.cells.forEach(function (c) { f.d.getRange(c.getRow(), 2, 1, 2).clearContent(); f.d.getRange(c.getRow(), 2).setBackground('#ffffff'); }); }

    }
    SpreadsheetApp.flush();
    P.setProperty('VER', String(upd)); CacheService.getScriptCache().put('ver', String(upd), 21600);
    v[2] = upd; v[10] = 'del';
  } finally { lock.releaseLock(); }

}

/* ================= связь со «списком дел» ================= */
function dela_() { return SpreadsheetApp.openById(DELA_SHEET_ID).getSheetByName(DELA_TAB); }
function parseDM_(x) {
  const m = String(x || '').trim().match(/^(\d{1,2})\s*[.,\s\/]\s*(\d{1,2})(?:\D|$)/);
  return m ? ('0' + m[1]).slice(-2) + '.' + ('0' + m[2]).slice(-2) : null;
}
// Та же логика, что в «Пульте»: строки только дописываются вниз, дата — в первой строке дня.
function addTask_(date, text) {
  const s = dela_(), last = s.getLastRow();
  const vals = last ? s.getRange(1, 1, last, 2).getDisplayValues() : [];
  const target = parseDM_(date);
  let lastDay = null;
  vals.forEach(function (v) { const dm = parseDM_(v[0]); if (dm) lastDay = dm; });
  let r;
  if (last && lastDay === target) r = last + 1;
  else { r = last ? last + 2 : 1; s.getRange(r, 1).setNumberFormat('@').setValue(date); }
  s.getRange(r, 2).setValue(text).setBackground('#ffffff');
  return r;
}
// Цвета «Пульта»: зелёный — решено, жёлтый — «не нужно» или перенос. Любой другой цвет (белый, ручная подсветка) — открыта.
function colorStatus_(hex) {
  const m = /^#?([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})/i.exec(hex || ''); if (!m) return 'open';
  const r = parseInt(m[1], 16) / 255, g = parseInt(m[2], 16) / 255, b = parseInt(m[3], 16) / 255;
  if (g > 0.55 && r < 0.5 && b < 0.6) return 'done';          // зелёный
  if (r > 0.8 && g > 0.75 && b < 0.45) return 'skip';          // жёлтый
  return 'open';
}

/** Смотрит, что помощник сделал с проблемами в «Пульте», и переносит статус в чат. */
function checkStatuses_() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return;
  try {
    const s = sh_('msgs'), last = s.getLastRow();
    if (last < 2) return;
    const from = Math.max(2, last - 3000 + 1);
    const vals = s.getRange(from, 1, last - from + 1, M_COLS.length).getValues();
    const open = {};
    vals.forEach(function (v, i) {
      if (String(v[4]) === 'prob' && v[9] && ['done', 'skip', 'gone', 'del'].indexOf(String(v[10])) < 0) open[v[9]] = i;
    });
    if (!Object.keys(open).length) return;

    const d = dela_(), dl = d.getLastRow();
    if (!dl) return;
    const df = 1, n = dl; // читаем весь «список дел»: иначе старые проблемы ошибочно получали «Убрано из дел»
    const dv = d.getRange(df, 1, n, 3).getDisplayValues(), bg = d.getRange(df, 2, n, 1).getBackgrounds();
    const found = {}; let cur = null;
    dv.forEach(function (v, i) {
      if (parseDM_(v[0])) cur = parseDM_(v[0]);
      const m = /\[(п\d+)\]\s*$/.exec(String(v[1]).trim());
      if (m && open.hasOwnProperty(m[1])) found[m[1]] = { st: colorStatus_(bg[i][0]), date: cur, comment: v[2] };
    });

    let changed = false;
    Object.keys(open).forEach(function (tag) {
      const i = open[tag], v = vals[i], f = found[tag];
      let st, note;
      if (!f) { st = 'gone'; note = ''; }
      else if (f.st === 'done') { st = 'done'; note = f.date || ''; }
      else if (f.st === 'skip') { st = /перенос/i.test(f.comment) ? 'open' : 'skip'; note = f.date || ''; }
      else { st = 'open'; note = f.date || ''; }
      if (st === v[10] && note === String(v[11])) return;
      if (!f && dl < 5) return; // защита от пустого чтения
      const upd = Math.max(Date.now(), verGet_() + 1);
      s.getRange(from + i, 3).setValue(upd);
      s.getRange(from + i, 11, 1, 2).setValues([[st, note]]);
      P.setProperty('VER', String(upd)); CacheService.getScriptCache().put('ver', String(upd), 21600);
      changed = true;
      if (st === 'done' || st === 'skip') {
        const txt = String(v[7] || 'см. фото');
        try { sendTo_([+v[5]].concat(adminIds_(String(v[3]))), st === 'done' ? '✅ Решено' : 'Проблема закрыта', BARS[v[3]] + ': ' + txt.slice(0, 120), APP_URL + '#prob-' + v[3]); } catch (e) {}
      }
    });
    if (changed) SpreadsheetApp.flush();
  } finally { lock.releaseLock(); }
}

const FCM_PROJECT = 'rabochiy-chat'; // проект Firebase

function sendTo_(uids, title, body, link) {
  const target = /#(prob|book)-(13|epo)$/.exec(String(link || ''));
  if (target) {
    const section = target[1] === 'prob' ? 'issues' : 'booking';
    const permitted = users_(true).filter(function (u) {
      return hasBarSection_(u, target[2], section, false);
    }).map(function (u) { return u.id; });
    uids = uids.filter(function (uid) { return permitted.indexOf(uid) >= 0; });
  }

  if (!uids.length) return;
  const s = sh_('push'), rows = rows_(s, ['uid', 'fcm', 'ts']).filter(function (r) { return uids.indexOf(+r.uid) >= 0 && r.fcm; });
  if (!rows.length) return;
  // Уведомления отправляются от имени владельца скрипта (он владелец проекта Firebase) — без файлов-ключей.
  const at = ScriptApp.getOAuthToken();
  const url = 'https://fcm.googleapis.com/v1/projects/' + FCM_PROJECT + '/messages:send';
  const reqs = rows.map(function (r) {
    return {
      url: url, method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + at, 'x-goog-user-project': FCM_PROJECT },
      payload: JSON.stringify({ message: {
        token: r.fcm,
        data: { title: title, body: body, link: link },
        webpush: { headers: { Urgency: 'high', TTL: '86400' }, fcm_options: { link: link },
          notification: { title: title, body: body, icon: APP_URL + 'icon-192.png', badge: APP_URL + 'icon-192.png' } }
      } })
    };
  });
  const res = UrlFetchApp.fetchAll(reqs);
  const dead = [];
  res.forEach(function (x, i) {
    const code = x.getResponseCode();
    if (code === 404 || /UNREGISTERED/.test(x.getContentText())) dead.push(rows[i].fcm);
    else if (code !== 200) console.warn('fcm ' + code + ' ' + x.getContentText().slice(0, 300));
  });
  if (!dead.length) return;
  // удаляем по самому токену и под блокировкой: номера строк могли сдвинуться, пока шла отправка
  const lock = LockService.getScriptLock(), held = lock.hasLock();
  if (!held && !lock.tryLock(10000)) return;
  try {
    rows_(s, ['uid', 'fcm', 'ts']).filter(function (r) { return dead.indexOf(r.fcm) >= 0; })
      .map(function (r) { return r._row; }).sort(function (a, b) { return b - a; }).forEach(function (r) { s.deleteRow(r); });
  } finally { if (!held) lock.releaseLock(); }
}

function bookSave_(me, x) {
  x = x || {};
  const bar = String(x.bar);
  requireBarSection_(me, bar, 'booking', true);
  const date = String(x.date || ''), tbl = String(x.tbl || '').trim(), time = String(x.time || '').trim();
  const name = String(x.name || '').trim().slice(0, 80), guests = String(x.guests || '').trim().slice(0, 10);
  const label = String(x.tlabel || '').trim().slice(0, 40) || tableName_(tbl);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) throw new Error('Нет даты');
  if (!tbl) throw new Error('Нет стола');
  if (!/^\d{1,2}:\d{2}$/.test(time)) throw new Error('Время в формате 19:30');
  if (!name) throw new Error('Нужно имя гостя');
  if (guests && !/^\d{1,3}$/.test(guests)) throw new Error('Гостей — числом');
  const s = sh_('bookings');
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  let isNew = false, res, cur = null;
  try {
    const all = bookRows_();
    cur = x.id ? all.filter(function (r) { return +r.id === +x.id && String(r.bar) === bar; })[0] : null;
    if (x.id && !cur) throw new Error('Бронь уже снята или изменена — обновите экран');
    if (date < todayIso_() && (!cur || String(cur.date) !== date)) throw new Error('Нельзя бронировать на прошедшую дату');
    if (all.some(function (r) { return String(r.bar) === bar && String(r.date) === date && String(r.tbl) === tbl && +r.id !== +x.id; }))
      throw new Error('Этот стол на эту дату уже забронирован');
    const vals = [bar, date, tbl, time, guests, name, String(x.phone || '').trim().slice(0, 30), String(x.note || '').trim().slice(0, 200)];
    if (cur) {
      s.getRange(cur._row, 2, 1, vals.length).setValues([vals]);
      res = bookPub_(rows_(s, B_COLS).filter(function (r) { return +r.id === +x.id; })[0]);
    } else {
      const id = rows_(s, B_COLS).reduce(function (m, r) { return Math.max(m, +r.id || 0); }, 0) + 1;
      s.appendRow([id].concat(vals).concat([me.id, me.name, new Date(), true]));
      isNew = true;
      res = bookPub_({ id: id, bar: bar, date: date, tbl: tbl, time: time, guests: guests, name: name, phone: vals[6], note: vals[7], uname: me.name, uid: me.id });
    }
  } finally { lock.releaseLock(); }
  try {
    const d = date.slice(8) + '.' + date.slice(5, 7);
    const line = d + ', ' + label + ', ' + time + (guests ? ', ' + guests + ' чел.' : '') + ' — ' + name;
    if (isNew) sendTo_(audience_(me, bar, 'book'), '📅 Новая бронь · ' + BARS[bar], line, APP_URL + '#book-' + bar);
    else if (cur && +cur.uid && +cur.uid !== me.id) sendTo_([+cur.uid], '✏️ Бронь изменена · ' + BARS[bar], me.name + ': ' + line, APP_URL + '#book-' + bar);
  } catch (e) { console.warn('push', e); }
  return res;
